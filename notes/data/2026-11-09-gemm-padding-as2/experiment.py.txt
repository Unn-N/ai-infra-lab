from pathlib import Path
import subprocess, json, hashlib, re, statistics, csv, io, shutil, sys

repo = Path('/home/unn/ai-infra-lab')
root = repo / 'build/padding_as2_2026-11-09'
oldroot = repo / 'build/padding_as_2026-11-07'
nvcc = '/usr/local/cuda/bin/nvcc'
ncu = '/usr/local/cuda/bin/ncu'
names = ['baseline', 'pad1', 'pad2']

def run(cmd, log):
    log.with_suffix('.command.json').write_text(json.dumps(cmd, indent=2))
    result = subprocess.run(cmd, capture_output=True, text=True)
    log.write_text(result.stdout + result.stderr)
    if result.returncode:
        print(result.stdout + result.stderr, flush=True)
        raise RuntimeError(f'Command failed: {cmd}')
    return result.stdout

phase = sys.argv[1]
if phase == 'prepare':
    root.mkdir(exist_ok=False)
    baseline = (repo/'gemm/gemm_self_re.cu').read_text()
    assert hashlib.sha256(baseline.encode()).hexdigest() == 'e0746da45b8a956068ca57632600a1b1c7ac4a3499ee2e2dc0b35218d20f5788'
    needle = '    __shared__ float As[OUTPUT_TILE][K_TILE];'
    assert baseline.count(needle) == 1
    sources = {'baseline': baseline,
               'pad1': (oldroot/'padA/gemm_self_re.cu').read_text(),
               'pad2': baseline.replace(needle, '    // [Codex +2 padding 实验] 有效 K 仍为32；行跨度136字节，保持8字节对齐。\n    __shared__ float As[OUTPUT_TILE][K_TILE + 2];')}
    metadata = {}
    for name, source in sources.items():
        folder = root/name
        folder.mkdir()
        (folder/'gemm_self_re.cu').write_text(source)
        for filename in ['gemm_self_bench.cu', 'gemm_self_check.cu']:
            shutil.copyfile(repo/'gemm'/filename, folder/filename)
        for filename in ['resource.cu', 'profile.cu']:
            shutil.copyfile(oldroot/'baseline'/filename, folder/filename)
        metadata[name] = {'sha256': hashlib.sha256(source.encode()).hexdigest(), 'As_stride': {'baseline':32,'pad1':33,'pad2':34}[name]}

    for name in names:
        folder = root/name
        for source, binary in [('gemm_self_bench.cu','bench'),('gemm_self_check.cu','check'),('resource.cu','resource'),('profile.cu','profile')]:
            run([nvcc,'-O3','-lineinfo','-arch=sm_120',str(folder/source),'-o',str(folder/binary)],folder/f'compile_{binary}.txt')
        print(f'Compiled {name}',flush=True)
        print(run([str(folder/'check')],root/f'{name}_check.txt'),flush=True)

        (root/f'{name}_resource.json').write_text(json.dumps(resource,indent=2))

        print(name,resource,counts,flush=True)
elif phase == 'measure':
    runs=[]

                parts=line.split()

            matches=[r['rows'][i] for r in runs if r['variant']==name]

        summary.append(entry)

## 假设及实际代码

As 每行136字节，保持8字节对齐；按32位标量访问，第一个 warp 的两组16线程读取相隔4行的地址，bank差从原版0变为8，+1则为4。该具体访问中两种padding都能错开，不能仅因步长34为偶数就断言其冲突改善必然弱于33。

候选保存为 gemm/gemm_self_pad2.cu。默认 gemm/gemm_self_re.cu 继续保留未padding基线；两份代码的有效计算相同。用户尚未要求提交或push。

## 结论与边界

这次验证了对齐推理：136B行跨度使宽加载恢复，同时保留明显的bank冲突改善。并未验证“恢复宽加载后必然明显加速”。原版存在冲突，并不代表消掉该项就能同比缩短总耗时。尚未唯一定位整个kernel的主要吞吐约束，也未将全部性能差异归因某一项指标。
