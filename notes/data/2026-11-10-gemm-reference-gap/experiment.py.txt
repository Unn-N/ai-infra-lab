def run(cmd, tag):
    (ROOT / (tag + '.command.json')).write_text(json.dumps(cmd, indent=2))
    proc = subprocess.run(cmd, capture_output=True, text=True)
    (ROOT / (tag + '.txt')).write_text(proc.stdout + proc.stderr)
    if proc.returncode:
        print((proc.stdout + proc.stderr)[-10000:], flush=True)
        raise RuntimeError((tag, proc.returncode))
    return proc.stdout

def save(name, value):
