from pathlib import Path
import subprocess, json, hashlib, re, statistics, csv, io, shutil, sys

repo=Path('/home/unn/ai-infra-lab')
root=repo/'build/const_block_2026-11-09'
previous=repo/'build/transpose_a_2026-11-09/transA'
nvcc='/usr/local/cuda/bin/nvcc'
names=['runtime','constant']

def run(cmd,path):
    path.with_suffix('.command.json').write_text(json.dumps(cmd,indent=2))
    r=subprocess.run(cmd,capture_output=True,text=True)
    path.write_text(r.stdout+r.stderr)
    if r.returncode:
        print(r.stdout+r.stderr,flush=True)
        raise RuntimeError(cmd)
    return r.stdout

phase=sys.argv[1]
if phase=='prepare':
    root.mkdir(exist_ok=False)
    baseline=(repo/'gemm/gemm_self_transpose_a.cu').read_text()
    assert hashlib.sha256(baseline.encode()).hexdigest()=='8a1cdb9617227d1b81f858df4f90607b0ec4657fc663f2cab84383207309be14'
    anchor='// ===== Codex 补充结束 ====='
    candidate=baseline.replace(anchor,'''// [Codex 编译期分工实验] 与当前启动方式一致，由输出分工推导固定线程形状。
constexpr int BLOCK_SIDE = OUTPUT_TILE / THREAD_TILE;  // 16
constexpr int BLOCK_THREADS = BLOCK_SIDE * BLOCK_SIDE; // 256
'''+anchor,1)
    old='    const int tid = ty * blockDim.x + tx;\n    const int threads = blockDim.x * blockDim.y;'
    assert candidate.count(old)==1
    candidate=candidate.replace(old,'''    // [Codex 编译期分工实验] 本kernel按BLOCK_SIDE×BLOCK_SIDE启动；让编译器看清搬运步长。
    const int tid = ty * BLOCK_SIDE + tx;''')
    assert candidate.count('index += threads')==2
    candidate=candidate.replace('index += threads','index += BLOCK_THREADS')
    launch='    dim3 block(OUTPUT_TILE / THREAD_TILE, OUTPUT_TILE / THREAD_TILE);'
    assert candidate.count(launch)==1
    candidate=candidate.replace(launch,'    dim3 block(BLOCK_SIDE, BLOCK_SIDE); // [Codex] 与编译期搬运分工保持一致。')
    metadata={}
    for name,source in [('runtime',baseline),('constant',candidate)]:
        folder=root/name; folder.mkdir()
        (folder/'gemm_self_re.cu').write_text(source)
        for filename in ['gemm_self_bench.cu','gemm_self_check.cu','resource.cu','profile.cu']:
            text=(previous/filename).read_text()
            if filename=='gemm_self_bench.cu':
                text=text.replace('constexpr int warmup = 5;','constexpr int warmup = 20;').replace('constexpr int repeats = 20;','constexpr int repeats = 100;')
            (folder/filename).write_text(text)
        metadata[name]={'sha256':hashlib.sha256(source.encode()).hexdigest(),'output_tile':64,'K_tile':32,'thread_tile':4,'block_side':16,'block_threads':256}
        for sourcefile,binary in [('gemm_self_bench.cu','bench'),('gemm_self_check.cu','check'),('resource.cu','resource'),('profile.cu','profile')]:
            run([nvcc,'-O3','-lineinfo','-arch=sm_120',str(folder/sourcefile),'-o',str(folder/binary)],folder/f'compile_{binary}.txt')
        print('Compiled',name,flush=True)
        print(run([str(folder/'check')],root/f'{name}_check.txt'),flush=True)
        resource=json.loads(run([str(folder/'resource')],root/f'{name}_resource.txt'))
        (root/f'{name}_resource.json').write_text(json.dumps(resource,indent=2))
        sass=run(['/usr/local/cuda/bin/cuobjdump','--dump-sass',str(folder/'bench')],root/f'{name}_sass.txt')
        opcodes=re.findall(r'/\*[0-9a-f]+\*/\s+(?:@!?[A-Z0-9]+\s+)?([A-Z][A-Z0-9]*(?:\.[A-Z0-9_]+)*)\s',sass)
        counts={key:opcodes.count(key) for key in sorted(set(opcodes))}
        (root/f'{name}_sass_counts.json').write_text(json.dumps(counts,indent=2))
        print(name,resource,'static instructions',sum(counts.values()),flush=True)
    (root/'metadata.json').write_text(json.dumps(metadata,indent=2))
elif phase=='measure':
    runs=[]; repeats={name:0 for name in names}
    for name in ['runtime','constant','constant','runtime','runtime','constant']:
        repeats[name]+=1
        output=run([str(root/name/'bench')],root/f'{name}_run{repeats[name]}.txt')
        rows=[]
        for line in output.splitlines():
            parts=line.split()
            if len(parts)==9 and all(x.isdigit() for x in parts[:3]):
                rows.append({'shape':list(map(int,parts[:3])),'rounds_ms':list(map(float,parts[3:6])),'median_ms':float(parts[6])})
        assert len(rows)==5,output
        result={'variant':name,'repeat':repeats[name],'rows':rows}
        runs.append(result); print(result,flush=True)
    summary=[]
    for i in range(5):
        entry={'shape':runs[0]['rows'][i]['shape']}
        for name in names:
            matches=[r['rows'][i] for r in runs if r['variant']==name]
            values=[v for row in matches for v in row['rounds_ms']]
            entry[name]={'median_ms':statistics.median(values),'all_round_ms':values,'process_medians':[row['median_ms'] for row in matches]}
        entry['speedup']=entry['runtime']['median_ms']/entry['constant']['median_ms']
        summary.append(entry)
    (root/'runs.json').write_text(json.dumps(runs,indent=2))
    (root/'summary.json').write_text(json.dumps(summary,indent=2))
    print(json.dumps(summary,indent=2),flush=True)
elif phase=='profile':

            run(cmd,root/f'{tag}_capture.txt')
            output=run(['/usr/local/cuda/bin/ncu','--import',str(root/f'{tag}.ncu-rep'),'--page','raw','--csv','--print-units','base'],root/f'{tag}.csv')

            result={key:float(row[key].replace(',','')) for key in metrics}
            result.update(variant=name,repeat=repeat)
            results.append(result); print(json.dumps(result),flush=True)

    for label,key,digits in [('动态warp指令总数','smsp__inst_executed.sum',0),('调度器发射活跃，active周期 %','smsp__issue_active.avg.pct_of_peak_sustained_active',2),('动态shared加载warp指令','smsp__sass_inst_executed_op_shared_ld.sum',0),('动态shared存储warp指令','smsp__sass_inst_executed_op_shared_st.sum',0)]:

    note='''# 28 · 固定线程分工改用编译期常量

2026-11-09晚间。用户提出编译期线程常量和float4搬运两项方向，明确授权现在只做前者。本轮以A局部转置、K32版为基线，输出64×64、每线程4×4、block16×16、Ats32×68、Bs32×64、单缓冲均保持。未实施float4、其他尺寸或布局修改。

## 实际改动

新增constexpr BLOCK_SIDE=OUTPUT_TILE/THREAD_TILE（16）、BLOCK_THREADS=BLOCK_SIDE*BLOCK_SIDE（256）。kernel内tid由ty*blockDim.x+tx改为ty*BLOCK_SIDE+tx；移除运行时threads变量，两段搬运循环都以BLOCK_THREADS为步长。host run_gemm的dim3 block也使用BLOCK_SIDE，保证启动与固定分工一致。外部入口当前仍由同一OUTPUT_TILE/THREAD_TILE表达式启动16×16，配置等价。

这版kernel的固定搬运分工要求按指定16×16启动；没有声称支持任意block尺寸。没有手工添加#pragma unroll，是否展开由编译器决定。
